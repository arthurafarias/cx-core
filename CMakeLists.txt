cmake_minimum_required(VERSION 3.17)
project(cxcore VERSION 0.2.0)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

include(CMakePackageConfigHelpers)
include(GNUInstallDirs)

# cx-core is the infrastructure shared by cx-flow, cx-networking and
# agenticx-ncortex, split by functionality: every libraries/cx-core-<namespace>
# is its own header-only INTERFACE target (namespace cx::core::<namespace>,
# headers under cx/core/<namespace>/) that can be linked on its own. Listed in
# dependency order - a library may only DEPENDS on one added before it.
set(CXCORE_LIBRARIES
  cx-core-testing
  cx-core-threading
  cx-core-signals
  cx-core-events
  cx-core-containers
  cx-core-io
)

foreach(library IN LISTS CXCORE_LIBRARIES)
  add_subdirectory(libraries/${library})
endforeach()

# Umbrella: everything at once, plus the pre-split <cxcore/...> include root,
# whose headers only forward to cx/core/<namespace>/ and re-export the flat
# cx::core names existing consumers were written against.
add_library(${PROJECT_NAME} INTERFACE)
add_library(${PROJECT_NAME}::${PROJECT_NAME} ALIAS ${PROJECT_NAME})

foreach(library IN LISTS CXCORE_LIBRARIES)
  target_link_libraries(${PROJECT_NAME} INTERFACE ${library}::${library})
endforeach()

target_include_directories(${PROJECT_NAME} INTERFACE
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
)

install(TARGETS ${PROJECT_NAME}
    EXPORT ${PROJECT_NAME}Targets
    INCLUDES DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}
)

install(DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/include/
    DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}
)

install(EXPORT ${PROJECT_NAME}Targets
    FILE ${PROJECT_NAME}Targets.cmake
    NAMESPACE ${PROJECT_NAME}::
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/${PROJECT_NAME}
)

write_basic_package_version_file(
    ${CMAKE_CURRENT_BINARY_DIR}/${PROJECT_NAME}ConfigVersion.cmake
    VERSION ${PROJECT_VERSION}
    COMPATIBILITY AnyNewerVersion
)

string(REPLACE ";" "\n" CXCORE_FIND_DEPENDENCIES "${CXCORE_LIBRARIES}")
string(REGEX REPLACE "(cx-core-[a-z]+)" "find_dependency(\\1)" CXCORE_FIND_DEPENDENCIES "${CXCORE_FIND_DEPENDENCIES}")

configure_package_config_file(
    ${CMAKE_CURRENT_SOURCE_DIR}/CMakeTemplates/Config.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/${PROJECT_NAME}Config.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/${PROJECT_NAME}
    PATH_VARS CMAKE_INSTALL_PREFIX
)

install(FILES
    ${CMAKE_CURRENT_BINARY_DIR}/${PROJECT_NAME}Config.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/${PROJECT_NAME}ConfigVersion.cmake
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/${PROJECT_NAME}
)

# Self-hosted test facility: no external test framework - the library stays
# dependency-free beyond the STL and pthreads. Each
# libraries/*/include/.../testing/*_test.hpp self-registers a test_group (see
# cx/core/testing/test_group.hpp); every one found is glob'd into a single
# generated .cpp of #includes and compiled into one test binary.
# CONFIGURE_DEPENDS re-runs the glob when a new one is added.
option(CXCORE_BUILD_TESTS "Build cxcore's self-hosted test facility" ON)
if(CXCORE_BUILD_TESTS)
  set(CXCORE_GENERATED_TESTS "${CMAKE_CURRENT_BINARY_DIR}/all_tests.cpp")
  file(WRITE "${CXCORE_GENERATED_TESTS}"
    "// Generated by CMake. Every testing/*_test.hpp header self-registers a test_group.\n")

  foreach(library IN LISTS CXCORE_LIBRARIES)
    set(LIBRARY_INCLUDE ${CMAKE_CURRENT_SOURCE_DIR}/libraries/${library}/include)
    file(GLOB_RECURSE LIBRARY_TEST_HEADERS CONFIGURE_DEPENDS ${LIBRARY_INCLUDE}/*_test.hpp)
    foreach(TEST_HEADER IN LISTS LIBRARY_TEST_HEADERS)
      file(RELATIVE_PATH RELATIVE_HEADER ${LIBRARY_INCLUDE} ${TEST_HEADER})
      file(APPEND "${CXCORE_GENERATED_TESTS}" "#include <${RELATIVE_HEADER}>\n")
    endforeach()
  endforeach()

  add_executable(${PROJECT_NAME}-tests
    tests/main.cpp
    tests/compat.cpp
    "${CXCORE_GENERATED_TESTS}"
  )
  target_link_libraries(${PROJECT_NAME}-tests PRIVATE ${PROJECT_NAME})

  enable_testing()
  add_test(NAME ${PROJECT_NAME}-tests COMMAND ${PROJECT_NAME}-tests)
endif()
